cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(picolearn C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(picolearn picolearn.c)
pico_enable_stdio_usb(picolearn 1)
pico_enable_stdio_uart(picolearn 0)
target_link_libraries(picolearn pico_stdlib)
pico_add_extra_outputs(picolearn)
