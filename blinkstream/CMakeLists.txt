cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(blinkstream C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(blinkstream blinkstream.c)
pico_enable_stdio_usb(blinkstream 1)
pico_enable_stdio_uart(blinkstream 0)
target_link_libraries(blinkstream pico_stdlib)
pico_add_extra_outputs(blinkstream)
